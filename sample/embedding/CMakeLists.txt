cmake_minimum_required(VERSION 3.25)
project(lhat_embedding_example LANGUAGES C)

set(LHAT_SOURCE_DIR "" CACHE PATH "L^ source tree; leave empty to use an installed SDK")
if(LHAT_SOURCE_DIR)
  set(LHAT_BUILD_TESTS OFF CACHE BOOL "")
  set(LHAT_BUILD_CLI OFF CACHE BOOL "")
  set(LHAT_BUILD_LSP OFF CACHE BOOL "")
  set(LHAT_BUILD_DAP OFF CACHE BOOL "")
  set(LHAT_BUILD_STDLIB OFF CACHE BOOL "")
  add_subdirectory("${LHAT_SOURCE_DIR}" lhat)
  set(embedding_libraries lhat lhatport)
else()
  find_package(lhat CONFIG REQUIRED)
  set(embedding_libraries lhat::lhat lhat::lhatport)
endif()
add_executable(embedding_host host.c)
target_compile_features(embedding_host PRIVATE c_std_11)
target_link_libraries(embedding_host PRIVATE ${embedding_libraries})

enable_testing()
add_test(NAME embedding_frames
  COMMAND embedding_host "${CMAKE_CURRENT_SOURCE_DIR}/main.lh")
add_test(NAME embedding_host_api COMMAND embedding_host --dump-host-api)
