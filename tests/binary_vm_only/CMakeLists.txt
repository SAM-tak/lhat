cmake_minimum_required(VERSION 3.25)
project(binary_vm_only LANGUAGES C)
set(LHAT_WITH_FRONTEND OFF CACHE BOOL "" FORCE)
set(LHAT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LHAT_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LHAT_BUILD_LSP OFF CACHE BOOL "" FORCE)
set(LHAT_BUILD_DAP OFF CACHE BOOL "" FORCE)
set(LHAT_WITH_DEBUGGER OFF CACHE BOOL "" FORCE)
set(LHAT_WITH_NET OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. lhat)
add_executable(binary_vm_only ../test_binary_data.c ../testutil.c)
target_include_directories(binary_vm_only PRIVATE ..)
target_link_libraries(binary_vm_only PRIVATE lhatstdlib)
enable_testing()
add_test(NAME binary_vm_only COMMAND binary_vm_only)
