# lex -- L^ 0.3.3 の英語から訳した。
# none = no error
none = 誤りなし
# unexpected-character = unexpected character
unexpected-character = 予期しない文字
# invalid-utf8 = invalid UTF-8 sequence
invalid-utf8 = UTF-8 として不正な並び
# bare-hat = '^' must directly follow an identifier
bare-hat = '^' は識別子の直後に書く
# bare-at = '@' must be followed directly by the name of an annotation
bare-at = '@' の直後には注釈の名前を書く
# ident-after-number = a number must be separated from an identifier by whitespace
ident-after-number = 数と識別子の間には空白が要る
# malformed-number = malformed number literal
malformed-number = 数の書き方が壊れている
# malformed-exponent = exponent must be followed by digits
malformed-exponent = 指数の後には数字が要る
# integer-overflow = integer literal is out of range
integer-overflow = 整数の字面が範囲を超えている
# unterminated-string = unterminated string literal
unterminated-string = 文字列が閉じていない
# unterminated-name-literal = name literal is not closed before the end of the line
unterminated-name-literal = 名前リテラルが行末までに閉じていない
# empty-name-literal = name literal is empty
empty-name-literal = 名前リテラルが空である
# unknown-escape = unknown escape sequence
unknown-escape = 知らないエスケープ
# malformed-escape = malformed escape sequence
malformed-escape = エスケープの書き方が壊れている
# unterminated-block-comment = unterminated block comment
unterminated-block-comment = ブロックコメントが閉じていない
# scope-without-name = scope specifier must be followed directly by a name
scope-without-name = スコープ指定子の直後には名前を書く
# interpolation-needs-quotes = string interpolation requires double quotes: $"..."
interpolation-needs-quotes = 文字列補間には二重引用符が要る: $"..."
# interpolation-too-deep = interpolated strings are nested too deeply
interpolation-too-deep = 補間した文字列の入れ子が深すぎる
