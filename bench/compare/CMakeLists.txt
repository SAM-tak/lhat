cmake_minimum_required(VERSION 3.25)
project(lhat_luau_lua_compare LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()

get_filename_component(LHAT_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LUAU_SOURCE_DIR "${LHAT_SOURCE_DIR}/build/compare-deps/luau" CACHE PATH "Explicitly prepared Luau checkout")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/lua.lock.json" lua_lock)
string(JSON lua_version GET "${lua_lock}" version)
string(JSON lua_archive_sha GET "${lua_lock}" sha256)
string(JSON lua_expected_source_sha GET "${lua_lock}" source_sha256)
set(LUA_SOURCE_DIR "${LHAT_SOURCE_DIR}/build/compare-deps/lua-${lua_version}" CACHE PATH "Explicitly prepared official Lua source")
# Match dependencies.lua_source_hash: fingerprint all C sources and headers,
# including luac. Reject modified or incomplete source trees before building.
file(GLOB lua_source_files RELATIVE "${LUA_SOURCE_DIR}" "${LUA_SOURCE_DIR}/src/*.c" "${LUA_SOURCE_DIR}/src/*.h")
list(SORT lua_source_files)
set(lua_source_entries "")
foreach(path IN LISTS lua_source_files)
    file(SHA256 "${LUA_SOURCE_DIR}/${path}" digest)
    string(APPEND lua_source_entries "${path}:${digest}\n")
endforeach()
string(SHA256 lua_source_sha "${lua_source_entries}")
if(NOT lua_source_sha STREQUAL lua_expected_source_sha)
    message(FATAL_ERROR "Lua source must match lua.lock.json; run prepare.py")
endif()
set(COMPARE_LUAU_OPT_LEVEL "2" CACHE STRING "Luau bytecode optimization level (0, 1, 2)")
if(NOT COMPARE_LUAU_OPT_LEVEL MATCHES "^[012]$")
    message(FATAL_ERROR "COMPARE_LUAU_OPT_LEVEL must be 0, 1, or 2")
endif()
find_package(Git REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/luau.lock.json" lock)
string(JSON pinned_sha GET "${lock}" commit)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${LUAU_SOURCE_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE luau_sha OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT luau_sha STREQUAL pinned_sha)
    message(FATAL_ERROR "Luau checkout must match luau.lock.json; run prepare.py")
endif()

# These are benchmark-only settings, scoped to this independent build tree.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LHAT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LHAT_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LHAT_BUILD_LSP OFF CACHE BOOL "" FORCE)
set(LHAT_BUILD_DAP OFF CACHE BOOL "" FORCE)
set(LHAT_WITH_DEBUGGER OFF CACHE BOOL "" FORCE)
set(LHAT_WITH_FRONTEND ON CACHE BOOL "" FORCE)
set(LHAT_BUILD_BENCH OFF CACHE BOOL "" FORCE)
set(LHAT_JIT OFF CACHE BOOL "" FORCE)
set(LHAT_SANITIZE OFF CACHE BOOL "" FORCE)
set(LHAT_PGO OFF CACHE STRING "" FORCE)
add_subdirectory("${LHAT_SOURCE_DIR}" lhat EXCLUDE_FROM_ALL)

set(LUAU_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_WEB OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(LUAU_EXTERN_C OFF CACHE BOOL "" FORCE)
set(LUAU_STATIC_CRT OFF CACHE BOOL "" FORCE)
add_subdirectory("${LUAU_SOURCE_DIR}" luau EXCLUDE_FROM_ALL)

# CORE_O plus lauxlib from the official Makefile. No standard libraries are
# needed by these cases; no changes are made to the upstream sources.
set(lua_core_sources lapi.c lcode.c lctype.c ldebug.c ldo.c ldump.c lfunc.c lgc.c
    llex.c lmem.c lobject.c lopcodes.c lparser.c lstate.c lstring.c ltable.c
    ltm.c lundump.c lvm.c lzio.c lauxlib.c)
list(TRANSFORM lua_core_sources PREPEND "${LUA_SOURCE_DIR}/src/")
add_library(lua_reference STATIC ${lua_core_sources})
target_include_directories(lua_reference PUBLIC "${LUA_SOURCE_DIR}/src")
if(NOT WIN32)
    target_link_libraries(lua_reference PUBLIC m)
endif()
add_executable(lua_bytecode "${LUA_SOURCE_DIR}/src/luac.c")
target_link_libraries(lua_bytecode PRIVATE lua_reference)
add_executable(lua_compare lua_runner.c)
target_link_libraries(lua_compare PRIVATE lua_reference)
add_dependencies(lua_compare lua_bytecode)

add_executable(lhat_compare lhat_runner.c)
target_link_libraries(lhat_compare PRIVATE lhat)
target_include_directories(lhat_compare PRIVATE "${LHAT_SOURCE_DIR}/src")
add_executable(luau_compare luau_runner.cpp)
target_link_libraries(luau_compare PRIVATE Luau.Compiler Luau.VM)
target_compile_definitions(luau_compare PRIVATE COMPARE_LUAU_OPT_LEVEL=${COMPARE_LUAU_OPT_LEVEL})
if(MSVC)
    target_compile_definitions(lua_compare PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(lua_compare PRIVATE /W4 /utf-8)
    target_compile_options(lhat_compare PRIVATE /W4 /utf-8)
    target_compile_options(luau_compare PRIVATE /W4 /utf-8)
else()
    target_compile_options(lua_compare PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(lhat_compare PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(luau_compare PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    target_link_libraries(lua_compare PRIVATE psapi)
    target_link_libraries(lhat_compare PRIVATE psapi)
    target_link_libraries(luau_compare PRIVATE psapi)
endif()

# The cache and compile_commands.json are also copied into measurement results.
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/compare-build-$<CONFIG>.json" CONTENT
"{\n  \"configuration\": \"$<CONFIG>\",\n  \"c_compiler\": \"${CMAKE_C_COMPILER_ID} ${CMAKE_C_COMPILER_VERSION}\",\n  \"cxx_compiler\": \"${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\",\n  \"luau_commit\": \"${luau_sha}\",\n  \"luau_optimization\": ${COMPARE_LUAU_OPT_LEVEL},\n  \"lua_version\": \"${lua_version}\",\n  \"lua_archive_sha256\": \"${lua_archive_sha}\",\n  \"lua_source_sha256\": \"${lua_source_sha}\",\n  \"jit\": false,\n  \"pgo\": false,\n  \"debugger\": false\n}\n")
